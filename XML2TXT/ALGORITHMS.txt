*** หลักการทำงาน *** 
  สร้างโครงสร้างข้อมูล ตามตัวอย่างข้างบน 
    function addColumn(file, string) {
      ถ้า string เป็น path# 
        ลบ # เหลือแค่ path 
        addColInner(file, path)
      ถ้า string เป็น path@attr 
        split เป็น path และ attr
        addColAttr(file, path, attr)
    }
    function addColInner(file, path) {
        hPath = hash(path);
        columns = file.get(columns);
        column = columns.get(hPath);
        if null == column {
            column = new  { attr:{}, inner:-1 }; 
            columns.add(hPath, column);  //เพิ่ม คอลั่มน์ใหม่ให้ พาท
        }
        column.get(inner).set(file.countCol++); //เซ็ตหมายเลขคอลั่มน์ ให้ inner
    }
    function addColAttr(file, path, attr) {
        hPath = hash(path);
        columns = file.get(columns);
        column = columns.get(hPath);
        if null == column {
            column = new  { attr:{}, inner:-1 }; 
            columns.add(hPath, column); //เพิ่ม คอลั่มน์ใหม่ให้ พาท
        }   
        hAttr = hash(attr)
        column.get(attr).add(hAttr, file.countCol++); //เพิ่ม attr พร้อมหมายเลขคอลั่มน์   
    }

    ถ้าเป็น EV_ATTR และ value เป็น empty ใส่ "1" เข้าไปใน file.asmblyLine[hMapAttr.get(hash("attrName"))]
